<!-- Fleek UI snippets: Alpine overlay patterns (Pines-style x-data).
     Surfaces use popover/card tokens so they track theme + dark mode.
     x-cloak hides them until Alpine boots; x-transition for entrances. -->

<!-- Dropdown menu: focus moves into the menu on open and back to the
     trigger on close (Escape, item click, click-outside). ArrowUp/Down
     cycles items; Tab closes and lets focus continue naturally. -->
<div
  class="relative"
  x-data="{
    open: false,
    openMenu() {
      this.open = true;
      this.$nextTick(() => this.items()[0]?.focus());
    },
    closeMenu() {
      this.open = false;
      this.$nextTick(() => this.$refs.trigger.focus());
    },
    items() {
      return [...this.$refs.menu.querySelectorAll('[role=menuitem]')];
    },
    moveFocus(delta) {
      const list = this.items();
      const i = Math.max(0, list.indexOf(document.activeElement));
      list[(i + delta + list.length) % list.length]?.focus();
    },
  }">
  <button
    type="button"
    class="fui-btn fui-btn-outline"
    x-ref="trigger"
    x-on:click="open ? closeMenu() : openMenu()"
    x-on:keydown.arrow-down.prevent="openMenu()"
    x-bind:aria-expanded="open"
    aria-haspopup="menu">
    Actions
  </button>
  <div
    class="border-border bg-popover text-popover-foreground absolute right-0 z-50 mt-1 w-44 rounded-md border p-1 shadow-md"
    role="menu"
    x-ref="menu"
    x-show="open"
    x-cloak
    x-transition.opacity.scale.90
    x-on:click.outside="open = false"
    x-on:keydown.escape.window="open && closeMenu()"
    x-on:keydown.tab="open = false"
    x-on:keydown.arrow-down.prevent="moveFocus(1)"
    x-on:keydown.arrow-up.prevent="moveFocus(-1)">
    <button
      type="button"
      class="hover:bg-muted focus-visible:bg-muted flex w-full items-center rounded-sm px-2 py-1.5 text-sm"
      role="menuitem"
      x-on:click="closeMenu()">
      Rename
    </button>
    <button
      type="button"
      class="hover:bg-muted focus-visible:bg-muted flex w-full items-center rounded-sm px-2 py-1.5 text-sm"
      role="menuitem"
      x-on:click="closeMenu()">
      Duplicate
    </button>
    <div class="border-border my-1 border-t" role="separator"></div>
    <button
      type="button"
      class="text-destructive hover:bg-destructive/10 focus-visible:bg-destructive/10 flex w-full items-center rounded-sm px-2 py-1.5 text-sm"
      role="menuitem"
      x-on:click="closeMenu()">
      Delete
    </button>
  </div>
</div>
